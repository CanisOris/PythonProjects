1. Object Use and Manipulation

Creating instances, reading/updating attributes, validating state with @property, and customizing object behavior with special ("dunder") methods like __repr__ and __eq__. 

class BankAccount:
    def __init__(self, owner: str, balance: float = 0.0) -> None:
        self.owner = owner
        self.balance = balance  # Uses the setter below

    @property
    def balance(self) -> float:
        return self._balance

    @balance.setter
    def balance(self, value: float) -> None:
        if value < 0:
            raise ValueError("Balance cannot be negative.")
        self._balance = round(value, 2)

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("Deposit amount must be positive.")
        self.balance += amount

    def withdraw(self, amount: float) -> bool:
        if amount <= 0 or amount > self.balance:
            return False
        self.balance -= amount
        return True

    def __repr__(self) -> str:
        return f"BankAccount(owner={self.owner!r}, balance={self.balance:.2f})"

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, BankAccount):
            return NotImplemented
        return self.owner == other.owner and self.balance == other.balance


# Instantiation and state manipulation
acct = BankAccount("Maya", 250.0)
acct.deposit(75.5)
withdrew = acct.withdraw(100.0)

print(acct)                   # BankAccount(owner='Maya', balance=225.50)
print("Withdrew:", withdrew)  # Withdrew: True

# Dynamic attribute inspection and manipulation
setattr(acct, "currency", "USD")
print("Currency:", getattr(acct, "currency"))            # Currency: USD
print("Has overdraft:", hasattr(acct, "overdraft_limit")) # Has overdraft: False

# Value equality via __eq__
same_acct = BankAccount("Maya", 225.5)
print("Equal accounts:", acct == same_acct)  # True

2. Encapsulation

Bundling internal state with the methods that manage it while restricting direct external mutation. Python signals visibility using naming conventions—a single leading underscore (_protected) for internal/subclass use and a double leading underscore (__private) for name-mangled attributes—paired with controlled getter/setter interfaces. 

class Vault:
    def __init__(self, owner: str, pin: str, initial_funds: float = 0.0) -> None:
        self.owner = owner                  # Public attribute
        self._audit_log: list[str] = []     # Protected convention (internal / subclass use)
        self.__pin = pin                    # Name-mangled private attribute (_Vault__pin)
        self.__balance = 0.0
        if initial_funds > 0:
            self.deposit(initial_funds)

    @property
    def balance(self) -> float:
        """Read-only public view of the internal balance."""
        return self.__balance

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("Deposit must be positive.")
        self.__balance += amount
        self._record(f"Deposited {amount:.2f}")

    def withdraw(self, amount: float, pin: str) -> bool:
        if not self.__verify_pin(pin):
            self._record("Failed withdrawal: invalid PIN")
            return False
        if amount <= 0 or amount > self.__balance:
            return False
        self.__balance -= amount
        self._record(f"Withdrew {amount:.2f}")
        return True

    def get_audit_log(self) -> tuple[str, ...]:
        # Return an immutable copy so callers cannot mutate internal history
        return tuple(self._audit_log)

    def _record(self, entry: str) -> None:
        self._audit_log.append(entry)

    def __verify_pin(self, pin: str) -> bool:
        return pin == self.__pin


vault = Vault("Jordan", pin="4821", initial_funds=500.0)
print("Balance:", vault.balance)                        # Balance: 500.0
print("Bad PIN:", vault.withdraw(100.0, pin="0000"))    # Bad PIN: False
print("Good PIN:", vault.withdraw(100.0, pin="4821"))   # Good PIN: True
print("Audit log:", vault.get_audit_log())

# Direct access to __pin or __balance raises AttributeError
try:
    print(vault.__pin)
except AttributeError as err:
    print("Blocked direct access:", err)

3. Inheritance

Deriving specialized classes from a base class, reusing parent initialization/logic with super(), extending or overriding behavior, and checking class relationships. 

import math


class Shape:
    def __init__(self, color: str = "black") -> None:
        self.color = color

    def area(self) -> float:
        raise NotImplementedError("Subclasses must implement area().")

    def describe(self) -> str:
        return f"A {self.color} {type(self).__name__.lower()}"


class Rectangle(Shape):
    def __init__(self, width: float, height: float, color: str = "blue") -> None:
        super().__init__(color=color)
        self.width = width
        self.height = height

    def area(self) -> float:
        return self.width * self.height

    def describe(self) -> str:
        base_desc = super().describe()
        return f"{base_desc} ({self.width}x{self.height})"


class Circle(Shape):
    def __init__(self, radius: float, color: str = "red") -> None:
        super().__init__(color=color)
        self.radius = radius

    def area(self) -> float:
        return math.pi * (self.radius ** 2)


rect = Rectangle(4, 5)
circ = Circle(3)

print(rect.describe(), "-> area:", rect.area())           # A blue rectangle (4x5) -> area: 20
print(circ.describe(), "-> area:", round(circ.area(), 2)) # A red circle -> area: 28.27

# Relationship checks
print(isinstance(rect, Rectangle))  # True
print(isinstance(rect, Shape))      # True
print(issubclass(Circle, Shape))    # True

4. Polymorphism

Calling the same operation or method on objects of different types and letting each object respond according to its own implementation—via duck typing, shared interfaces, or operator overloading. 

import json


# A. Polymorphism via shared method signature (duck typing)
class EmailNotifier:
    def send(self, recipient: str, message: str) -> str:
        return f"[Email -> {recipient}] {message}"


class SmsNotifier:
    def send(self, recipient: str, message: str) -> str:
        return f"[SMS -> {recipient}] {message}"


class WebhookNotifier:
    def send(self, recipient: str, message: str) -> str:
        payload = json.dumps({"to": recipient, "msg": message})
        return f"[Webhook] {payload}"


def notify_all(notifiers: list, recipient: str, message: str) -> None:
    # Works with any object that defines a compatible .send(recipient, message) method
    for notifier in notifiers:
        print(notifier.send(recipient, message))


channels = [EmailNotifier(), SmsNotifier(), WebhookNotifier()]
notify_all(channels, "alex@example.com", "Deployment succeeded.")


# B. Built-in / operator polymorphism (len(), +)
class Playlist:
    def __init__(self, tracks: list[str]) -> None:
        self.tracks = list(tracks)

    def __len__(self) -> int:
        return len(self.tracks)

    def __add__(self, other: "Playlist") -> "Playlist":
        if not isinstance(other, Playlist):
            return NotImplemented
        return Playlist(self.tracks + other.tracks)


p1 = Playlist(["Song A", "Song B"])
p2 = Playlist(["Song C"])
combined = p1 + p2

print("Track count:", len(combined))  # Track count: 3
print("Tracks:", combined.tracks)     # Tracks: ['Song A', 'Song B', 'Song C']

5. Abstraction

Hiding low-level implementation details behind a clean, minimal contract so callers interact only with what an object does rather than how it does it. Python enforces abstract contracts using abc.ABC and @abstractmethod, preventing incomplete subclasses from being instantiated. 

from abc import ABC, abstractmethod


class PaymentProcessor(ABC):
    """Abstract contract defining the interface every payment gateway must provide."""

    @abstractmethod
    def authorize(self, amount: float) -> str:
        """Reserve funds and return an authorization token."""

    @abstractmethod
    def capture(self, auth_token: str) -> bool:
        """Finalize a previously authorized charge."""

    def process_payment(self, amount: float) -> bool:
        """High-level workflow built purely on abstract operations."""
        token = self.authorize(amount)
        return self.capture(token)


class StripeProcessor(PaymentProcessor):
    def authorize(self, amount: float) -> str:
        print(f"[Stripe] Authorizing ${amount:.2f} via Stripe API...")
        return "stripe_tok_9182"

    def capture(self, auth_token: str) -> bool:
        print(f"[Stripe] Capturing token {auth_token}")
        return True


class LocalMockProcessor(PaymentProcessor):
    def authorize(self, amount: float) -> str:
        return f"mock_tok_{int(amount * 100)}"

    def capture(self, auth_token: str) -> bool:
        return auth_token.startswith("mock_tok_")


def checkout(processor: PaymentProcessor, total: float) -> None:
    # Caller depends only on the PaymentProcessor abstraction, not gateway internals
    success = processor.process_payment(total)
    print("Checkout complete:", success)


checkout(StripeProcessor(), 49.99)
checkout(LocalMockProcessor(), 15.00)

# Attempting to instantiate an abstract base class directly raises TypeError
try:
    PaymentProcessor()
except TypeError as err:
    print("Cannot instantiate ABC:", err)

6. Terms and Commands Used

Definitions of core object-oriented concepts, naming conventions, built-in functions, decorators, and special ("dunder") methods demonstrated across the examples above.

Core OOP & Python Concepts

Class & Instance
A class is a blueprint defining attributes and methods; an instance is an individual object created by calling the class (e.g., BankAccount("Maya", 250.0)).
self
The conventional first parameter of instance methods, referring to the specific object instance on which the method was called.
Protected Convention (_name)
A single leading underscore signaling to developers that an attribute or helper method is intended for internal class or subclass use rather than public callers.
Name Mangling / Private Convention (__name)
A double leading underscore (without trailing double underscores) that Python rewrites internally to _ClassName__name to prevent accidental external access or subclass collisions.
Duck Typing
A style of polymorphism where an object's suitability is determined by the presence of expected methods and attributes (e.g., .send()) rather than its inheritance hierarchy.
Operator Overloading
Implementing special methods such as __add__ or __eq__ so custom objects work naturally with standard Python operators (+, ==).

Decorators & Standard Library Tools

@property / @<attr>.setter
Built-in decorators that expose methods as managed attributes, allowing validation or computed reads while preserving standard dot-access syntax (acct.balance).
abc.ABC & @abstractmethod
Base class and decorator from the abc (Abstract Base Classes) module used to define formal interfaces; Python raises TypeError if a subclass fails to implement all abstract methods.

Built-in Commands & Functions

super()
Returns a proxy object delegating method calls to a parent or sibling class, commonly used in __init__ or overridden methods to reuse base-class logic.
isinstance(obj, cls)
Returns True if obj is an instance of cls or any subclass thereof.
issubclass(sub, sup)
Returns True if class sub is derived from class sup.
getattr(obj, name) / setattr(obj, name, val) / hasattr(obj, name)
Built-in reflection functions for dynamically reading, assigning, or checking the existence of an object's attribute by string name.
type(obj).__name__
Retrieves the runtime class of obj and accesses its string class name.
NotImplemented & NotImplementedError
NotImplemented is a special return value signaling that a binary operation (like == or +) is unsupported for the given operand type; NotImplementedError is an exception raised when an abstract method has not been overridden.

Special ("Dunder") Methods

__init__(self, ...)
Initializer method automatically invoked when a new instance is created to set up its initial state.
__repr__(self)
Returns an unambiguous string representation of an object, used by repr(), interactive consoles, and fallback print() formatting.
__eq__(self, other)
Defines value equality behavior for the == operator.
__len__(self)
Defines how the built-in len(obj) function computes the size of an object.
__add__(self, other)
Defines addition behavior for the + operator.
