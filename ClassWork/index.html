Python 3 Object-Oriented Programming Examples

Runnable examples covering object use and manipulation, inheritance, and polymorphism.

1. Object Use and Manipulation

Creating instances, reading/updating attributes, encapsulating validation with @property, and customizing object behavior with special ("dunder") methods like __repr__ and __eq__. 

class BankAccount:
    def __init__(self, owner: str, balance: float = 0.0) -> None:
        self.owner = owner
        self.balance = balance  # Uses the setter below

    @property
    def balance(self) -> float:
        return self._balance

    @balance.setter
    def balance(self, value: float) -> None:
        if value < 0:
            raise ValueError("Balance cannot be negative.")
        self._balance = round(value, 2)

    def deposit(self, amount: float) -> None:
        if amount <= 0:
            raise ValueError("Deposit amount must be positive.")
        self.balance += amount

    def withdraw(self, amount: float) -> bool:
        if amount <= 0 or amount > self.balance:
            return False
        self.balance -= amount
        return True

    def __repr__(self) -> str:
        return f"BankAccount(owner={self.owner!r}, balance={self.balance:.2f})"

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, BankAccount):
            return NotImplemented
        return self.owner == other.owner and self.balance == other.balance


# Instantiation and state manipulation
acct = BankAccount("Maya", 250.0)
acct.deposit(75.5)
withdrew = acct.withdraw(100.0)

print(acct)                   # BankAccount(owner='Maya', balance=225.50)
print("Withdrew:", withdrew)  # Withdrew: True

# Dynamic attribute inspection and manipulation
setattr(acct, "currency", "USD")
print("Currency:", getattr(acct, "currency"))            # Currency: USD
print("Has overdraft:", hasattr(acct, "overdraft_limit")) # Has overdraft: False

# Value equality via __eq__
same_acct = BankAccount("Maya", 225.5)
print("Equal accounts:", acct == same_acct)  # True

2. Inheritance

Deriving specialized classes from a base class, reusing parent initialization/logic with super(), extending or overriding behavior, and checking class relationships. 

import math


class Shape:
    def __init__(self, color: str = "black") -> None:
        self.color = color

    def area(self) -> float:
        raise NotImplementedError("Subclasses must implement area().")

    def describe(self) -> str:
        return f"A {self.color} {type(self).__name__.lower()}"


class Rectangle(Shape):
    def __init__(self, width: float, height: float, color: str = "blue") -> None:
        super().__init__(color=color)
        self.width = width
        self.height = height

    def area(self) -> float:
        return self.width * self.height

    def describe(self) -> str:
        base_desc = super().describe()
        return f"{base_desc} ({self.width}x{self.height})"


class Circle(Shape):
    def __init__(self, radius: float, color: str = "red") -> None:
        super().__init__(color=color)
        self.radius = radius

    def area(self) -> float:
        return math.pi * (self.radius ** 2)


rect = Rectangle(4, 5)
circ = Circle(3)

print(rect.describe(), "-> area:", rect.area())           # A blue rectangle (4x5) -> area: 20
print(circ.describe(), "-> area:", round(circ.area(), 2)) # A red circle -> area: 28.27

# Relationship checks
print(isinstance(rect, Rectangle))  # True
print(isinstance(rect, Shape))      # True
print(issubclass(Circle, Shape))    # True

3. Polymorphism

Calling the same operation or method on objects of different types and letting each object respond according to its own implementation—via duck typing, shared interfaces, or operator overloading. 

import json


# A. Polymorphism via shared method signature (duck typing)
class EmailNotifier:
    def send(self, recipient: str, message: str) -> str:
        return f"[Email -> {recipient}] {message}"


class SmsNotifier:
    def send(self, recipient: str, message: str) -> str:
        return f"[SMS -> {recipient}] {message}"


class WebhookNotifier:
    def send(self, recipient: str, message: str) -> str:
        payload = json.dumps({"to": recipient, "msg": message})
        return f"[Webhook] {payload}"


def notify_all(notifiers: list, recipient: str, message: str) -> None:
    # Works with any object that defines a compatible .send(recipient, message) method
    for notifier in notifiers:
        print(notifier.send(recipient, message))


channels = [EmailNotifier(), SmsNotifier(), WebhookNotifier()]
notify_all(channels, "alex@example.com", "Deployment succeeded.")


# B. Built-in / operator polymorphism (len(), +)
class Playlist:
    def __init__(self, tracks: list[str]) -> None:
        self.tracks = list(tracks)

    def __len__(self) -> int:
        return len(self.tracks)

    def __add__(self, other: "Playlist") -> "Playlist":
        if not isinstance(other, Playlist):
            return NotImplemented
        return Playlist(self.tracks + other.tracks)


p1 = Playlist(["Song A", "Song B"])
p2 = Playlist(["Song C"])
combined = p1 + p2

print("Track count:", len(combined))  # Track count: 3
print("Tracks:", combined.tracks)     # Tracks: ['Song A', 'Song B', 'Song C']
